Откройте **Query Tool** в pgAdmin, вставьте этот скрипт и нажмите `F5` (Run).

```SQL
-- 1. Очистка старых таблиц (если есть)
DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS users;
DROP TABLE IF EXISTS departments;

-- 2. Создание структуры
CREATE TABLE departments (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    department_id INT REFERENCES departments(id),
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    balance NUMERIC(10, 2) DEFAULT 100.00,
    version INT DEFAULT 1 -- Поле для Optimistic Lock
);

CREATE TABLE orders (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    amount NUMERIC(10, 2) NOT NULL,
    status VARCHAR(20) DEFAULT 'CREATED'
);

-- 3. ГЕНЕРАЦИЯ ДАННЫХ (Магия функции generate_series)

-- Создаем 10 департаментов
INSERT INTO departments (name)
SELECT 'Department ' || i 
FROM generate_series(1, 10) i;

-- Создаем 100 000 пользователей со случайным балансом и случайным департаментом
INSERT INTO users (department_id, name, email, balance)
SELECT 
    (random() * 9 + 1)::int,                -- random id от 1 до 10
    'User ' || i,                           -- User 1, User 2...
    'user' || i || '@example.com', 
    (random() * 1000)::numeric(10,2)        -- Баланс от 0 до 1000
FROM generate_series(1, 100000) i;

-- Создаем 500 000 заказов, привязанных к случайным пользователям
INSERT INTO orders (user_id, amount)
SELECT 
    (random() * 99999 + 1)::bigint,         -- random user_id от 1 до 100 000
    (random() * 5000)::numeric(10,2)
FROM generate_series(1, 500000) i;

-- Создаем индексы для будущих тестов
CREATE INDEX idx_users_department_id ON users(department_id);
CREATE INDEX idx_orders_user_id ON orders(user_id);

-- Обновляем системную статистику планировщика!
ANALYZE departments;
ANALYZE users;
ANALYZE orders;
```
